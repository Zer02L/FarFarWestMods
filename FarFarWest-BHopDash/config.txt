# BHopDash Enhanced v13.0 Configuration for Far Far West
ENABLED=true

# Режимы работы:
# ALWAYS_BHOP=false (по умолчанию):
# Bhop работает ТОЛЬКО при зажатом Пробеле.
# При отпускании Пробела персонаж сразу прекращает прыгать при касании земли.
# Одиночное нажатие Пробела делает один обычный одиночный прыжок.
#
# ALWAYS_BHOP=true:
# Персонаж выпрыгивает при абсолютно каждом касании земли всегда, даже без зажатия клавиши.
ALWAYS_BHOP=false

# Окно удержания клавиши (в секундах, по умолчанию 0.30):
# Время между повторами нажатия клавиши от ОС (key-repeat rate).
# При зажатом Пробеле повторы приходят каждые 0.03-0.05 сек.
# Если Пробел отпущен, повторы прекращаются и bhop останавливается при приземлении.
SPACE_HOLD_WINDOW=0.30

# Dash-Hop:
# Нажатие/зажатие Shift активирует рывок с последующим прыжком.
AUTO_DASH_ENABLED=true
AUTO_JUMP_AFTER_DASH=true
RESET_DASH_COOLDOWN=true

# Частота опроса физики (мс)
POLL_MS=10

DEBUG=true
